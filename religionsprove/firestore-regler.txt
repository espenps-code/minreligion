// ===== Religionsprøven – lim inn INNI  match /databases/{database}/documents { ... }
// ===== i Firestore → Regler (krle-sim). Ikke fjern reglene for «klasser» (Krøniken).

    match /prover/{pid} {
      function prove() { return get(/databases/$(database)/documents/prover/$(pid)).data; }
      function erEier() { return request.auth != null && prove().eier == request.auth.uid; }

      allow get: if true;
      allow list: if request.auth != null && resource.data.eier == request.auth.uid;
      allow create: if request.auth != null && request.resource.data.eier == request.auth.uid;
      allow update, delete: if request.auth != null && resource.data.eier == request.auth.uid;

      match /koder/{kode} {
        allow get: if true;               // eleven kan sjekke én kode, men ikke liste alle
        allow list, write: if erEier();
      }

      match /svar/{kode} {
        allow get: if erEier() || resource == null || resource.data.levert == false;
        allow list, delete: if erEier();
        allow create: if prove().aktiv == true
          && exists(/databases/$(database)/documents/prover/$(pid)/koder/$(kode))
          && request.resource.data.kode == kode
          && request.resource.data.levert == false
          && request.resource.data.sporsmal.size() == 10;
        allow update: if prove().aktiv == true
          && resource.data.levert == false
          && request.resource.data.kode == resource.data.kode
          && request.resource.data.religion == resource.data.religion
          && request.resource.data.sporsmal == resource.data.sporsmal;
      }
    }
